    function selectFlavor(product, flavor, buttonEl) {

        btn.classList.remove('active', 'border-brand-primary', 'bg-brand-light', 'text-brand-primary', 'border-brand-accent', 'bg-amber-50', 'text-brand-accent');
        btn.classList.add('border-slate-200', 'bg-white', 'text-slate-700');
      });

      if (product === 'unimate') {
        buttonEl.classList.add('active', 'border-brand-primary', 'bg-brand-light', 'text-brand-primary');
      } else {
        buttonEl.classList.add('active', 'border-brand-accent', 'bg-amber-50', 'text-brand-accent');
      }

      // Mise à jour du résumé
      document.getElementById('pack-summary').innerHTML = 
        `Unimate : <strong>${selectedState.unimate}</strong> + Balance : <strong>${selectedState.balance}</strong>`;
    }

    // Gestion du Chatbot
    const chatWindow = document.getElementById('chat-window');
    const chatMessages = document.getElementById('chat-messages');
    const chatInput = document.getElementById('chat-input');

    function toggleChat(forceOpen) {
      if (typeof forceOpen === 'boolean') {
        chatWindow.classList.toggle('hidden', !forceOpen);
      } else {
        chatWindow.classList.toggle('hidden');
      }
      if (!chatWindow.classList.contains('hidden')) {
        chatInput.focus();
      }
    }

    function appendMessage(sender, text, isHtml = false) {
      const msgWrapper = document.createElement('div');
      msgWrapper.className = sender === 'user' ? 'flex justify-end' : 'flex items-start gap-2';

      if (sender === 'user') {
        msgWrapper.innerHTML = `
          <div class="bg-brand-primary text-white p-3 rounded-2xl rounded-tr-none text-slate-100 max-w-[85%] leading-relaxed shadow-sm">
            ${text}
          </div>
        `;
      } else {
        msgWrapper.innerHTML = `
          <div class="w-7 h-7 rounded-full bg-brand-primary text-white flex-shrink-0 flex items-center justify-center text-[10px] font-bold">FG</div>
          <div class="bg-white p-3 rounded-2xl rounded-tl-none border border-slate-200 shadow-sm text-slate-700 leading-relaxed max-w-[85%]">
            ${isHtml ? text : text}
          </div>
        `;
      }

      chatMessages.appendChild(msgWrapper);
      chatMessages.scrollTop = chatMessages.scrollHeight;
    }

    function sendQuickReply(text) {
      appendMessage('user', text);
      botAnswer(text);
    }

    function handleUserChat(e) {
      e.preventDefault();
      const text = chatInput.value.trim();
      if (!text) return;
      appendMessage('user', text);
      chatInput.value = '';
      botAnswer(text);
    }
